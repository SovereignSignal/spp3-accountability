# SPP3 accountability cron: TEMPLATE, not an installable crontab.
#
# The committee host's crontab is shared with jobs that are not part of this
# repository, and `crontab <file>` replaces the WHOLE user crontab. Never
# install this file directly. Merge these lines into `crontab -l` on the host,
# replacing WORKSPACE with the path of this checkout, and keep the merged file
# on the host, outside the repository.
#
#   crontab -l > /tmp/current && $EDITOR /tmp/current && crontab /tmp/current

MAILTO=""

# Daily stream health at 15:00 UTC. Alerts only on change.
0 15 * * * WORKSPACE/scripts/stream_monitor.py >> WORKSPACE/.cron-logs/streams.log 2>&1

# Weekly heartbeat, Mondays 15:05 UTC. Silence from a dead monitor and silence
# from a healthy system are otherwise indistinguishable.
5 15 * * 1 WORKSPACE/scripts/stream_monitor.py --heartbeat >> WORKSPACE/.cron-logs/streams.log 2>&1

# Quarterly report watcher, daily 15:10 UTC. Polls each provider's public forum
# thread, records reports that have landed, and nudges the committee privately
# before a due date passes. Never posts anywhere public.
10 15 * * * WORKSPACE/scripts/report_watcher.py >> WORKSPACE/.cron-logs/reports.log 2>&1

# Committee pipeline export, daily 15:20 UTC. Field whitelist; individual
# scores and contact details never leave Notion.
20 15 * * * WORKSPACE/scripts/notion_export.py >> WORKSPACE/.cron-logs/notion.log 2>&1
